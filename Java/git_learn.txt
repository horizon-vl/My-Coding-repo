git status (to see the status of the repository)
git add --all (to add all files to the staging area)
git add <file> (to add a file to the staging area)
git commit -m "" (to commit changes)
git restore <file> (to restore a file to its previous state)
git mv <file> <new_path> (to move a file) (also to rename a file)
git log (to see the log)
git log -p (to see the diff of a commit)
git commit "message" --amend (to amend the last commit)
git log --oneline (to see a concise log)
git reset <id> (to reset to a specific commit)
git rebase -i --roots (to interactively rebase)
git branch <branch_name> (to create a branch)
git branch (to list all branches)
git switch <branch_name> (to switch to a branch)
git merge -m "message" <branch_name>
git branch -d <branch_name> (to delete a branch)
git switch -c <branch_name> (to create and switch to a new branch)
git remote add origin <repository_url> (to add a remote repository)
git branch -M <branch_name> (to set it as the default branch)
git push -u origin <branch_name> (to push the branch to the remote repository)
git push --all (to push all branches to the remote repository)
git fetch (to fetch changes from the remote repository)
git pull (to pull changes from the remote repository)
